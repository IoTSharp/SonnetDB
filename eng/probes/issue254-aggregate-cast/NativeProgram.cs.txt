void Exec(string sql) { command.CommandText = sql; command.ExecuteNonQuery(); }
void Rows(string sql, Type[] expectedTypes, bool?[]? expectedNullable, params object[][] expected)
{
    Console.WriteLine("SQL|" + sql);
    command.CommandText = sql;
    using var reader = command.ExecuteReader();
    Console.WriteLine("METADATA|" + string.Join("|", Enumerable.Range(0, reader.FieldCount).Select(i => $"{reader.GetName(i)}:{reader.GetFieldType(i).Name}")));
    Require(expectedTypes.SequenceEqual(Enumerable.Range(0, reader.FieldCount).Select(reader.GetFieldType)), "Declared CLR types differ before Read.");
    if (expectedNullable is not null) Require(expectedNullable.SequenceEqual(reader.GetColumnSchema().Select(c => c.AllowDBNull)), "Nullability differs before Read.");
    int index = 0;
    while (reader.Read())
    {
        object[] values = Enumerable.Range(0, reader.FieldCount).Select(reader.GetValue).ToArray();
        Console.WriteLine("ROW|" + string.Join("|", values.Select(v => $"{v.GetType().Name}:{v}")));
        Require(index < expected.Length && expected[index].SequenceEqual(values), "Exact row values or runtime types differ.");
        for (int column = 0; column < values.Length; column++)
            if (expectedTypes[column] == typeof(double) && values[column] != DBNull.Value)
                Require(reader.GetFieldValue<double>(column).Equals(expected[index][column]) && reader.GetDouble(column).Equals(expected[index][column]), "Double typed getter mismatch.");
        index++;
    }
    Require(index == expected.Length, "Row count differs.");
}
void Error(string sql, string expectedMessage)
{
    command.CommandText = sql;
    Console.WriteLine("SQL|" + sql);
    try { using var reader = command.ExecuteReader(); while (reader.Read()) { } }
    catch (Exception exception)
    {
        Console.WriteLine($"EXPECTED-ERROR|{exception.GetType().Name}|{exception.Message}");
        Require(exception.ToString().Contains(expectedMessage, StringComparison.Ordinal), "Overflow stage or message differs.");
        return;
    }
    throw new InvalidOperationException("Expected overflow was accepted.");
}
void Run(string name, Action action)
{
    try { action(); passed++; Console.WriteLine($"PASS|{protocol}|{name}"); }
    catch (Exception exception) { failed++; Console.WriteLine($"FAIL|{protocol}|{name}|{exception.GetType().Name}|{exception.Message}"); }
}
void Require(bool condition, string message) { if (!condition) throw new InvalidOperationException(message); }
