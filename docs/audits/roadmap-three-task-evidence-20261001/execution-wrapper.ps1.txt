param([Parameter(Mandatory)][string]$ConfigPath)
$ErrorActionPreference = 'Stop'
if ($PSVersionTable.PSVersion.Major -lt 7) { throw 'PowerShell 7 required.' }
$config = Get-Content -LiteralPath $ConfigPath -Raw | ConvertFrom-Json
$records = [System.Collections.Generic.Dictionary[int,object]]::new()
$runWatch = [System.Diagnostics.Stopwatch]::StartNew()
$process = [System.Diagnostics.Process]::new()
$start = [System.Diagnostics.ProcessStartInfo]::new()
$start.FileName = $config.Executable
$start.WorkingDirectory = $config.WorkingDirectory
$start.UseShellExecute = $false
$start.CreateNoWindow = $true
$start.RedirectStandardOutput = $true
$start.RedirectStandardError = $true
foreach ($argument in $config.Arguments) { [void]$start.ArgumentList.Add([string]$argument) }
$start.Environment['DOTNET_CLI_TELEMETRY_OPTOUT'] = '1'
$start.Environment['DOTNET_NOLOGO'] = '1'
$start.Environment['MSBUILDDISABLENODEREUSE'] = '1'
if ($config.TempDirectory) {
    $start.Environment['TEMP'] = [string]$config.TempDirectory
    $start.Environment['TMP'] = [string]$config.TempDirectory
}
$process.StartInfo = $start
$exitCode = -1
$timeout = $false
$outTask = $null
$errTask = $null
$cleanup = [System.Collections.Generic.List[object]]::new()
try {
    [void]$process.Start()
    $rootPid = $process.Id
    $rootRecord = Get-CimInstance Win32_Process -Filter "ProcessId=$rootPid"
    if ($null -eq $rootRecord) {
        if (-not $process.HasExited) { throw 'Missing live root process identity.' }
        $rootRecord = [pscustomobject]@{ProcessId=$rootPid;ParentProcessId=$PID;CreationDate=$process.StartTime;CommandLine=($config.Executable + ' ' + ($config.Arguments -join ' '))}
    }
    $records[$rootPid] = $rootRecord
    $outTask = $process.StandardOutput.ReadToEndAsync()
    $errTask = $process.StandardError.ReadToEndAsync()
    Write-Output ('START {0} PID={1} timeout={2}s' -f $config.Name,$rootPid,$config.TimeoutSeconds)
    $maxChecks = [int][Math]::Ceiling($config.TimeoutSeconds / 3.0) + 2
    for ($check = 0; $check -lt $maxChecks -and $runWatch.Elapsed.TotalSeconds -lt $config.TimeoutSeconds; $check++) {
        $snapshot = @(Get-CimInstance Win32_Process)
        for ($depth = 0; $depth -lt 16; $depth++) {
            $added = 0
            foreach ($candidate in $snapshot) {
                if ($records.ContainsKey([int]$candidate.ParentProcessId) -and -not $records.ContainsKey([int]$candidate.ProcessId)) {
                    $parentRecord = $records[[int]$candidate.ParentProcessId]
                    if ($candidate.CreationDate -ge $parentRecord.CreationDate) {
                        $records[[int]$candidate.ProcessId] = $candidate
                        $added++
                    }
                }
            }
            if ($added -eq 0) { break }
        }
        if ($process.HasExited) { $exitCode = $process.ExitCode; break }
        if ($check % 10 -eq 0) { Write-Output ('PROGRESS {0} elapsed={1:N0}s processes={2}' -f $config.Name,$runWatch.Elapsed.TotalSeconds,$records.Count) }
        [void]$process.WaitForExit(3000)
    }
    if (-not $process.HasExited) { $timeout = $true; throw ('Timeout: ' + $config.Name) }
    $exitCode = $process.ExitCode
}
finally {
    $identityList = @($records.Values | Sort-Object CreationDate -Descending)
    foreach ($identity in $identityList) {
        $current = Get-CimInstance Win32_Process -Filter ('ProcessId=' + $identity.ProcessId)
        if ($null -eq $current) { continue }
        if ($current.CreationDate -eq $identity.CreationDate -and $current.CommandLine -eq $identity.CommandLine -and $current.ParentProcessId -eq $identity.ParentProcessId) {
            $cleanup.Add([pscustomobject]@{ProcessId=$current.ProcessId;CreationDate=$current.CreationDate;CommandLine=$current.CommandLine;ParentProcessId=$current.ParentProcessId})
            Stop-Process -Id $current.ProcessId -Force -ErrorAction SilentlyContinue
        }
    }
    if ($null -ne $outTask) { [System.IO.File]::WriteAllText($config.OutputPrefix + '.out.log', $outTask.GetAwaiter().GetResult()) }
    if ($null -ne $errTask) { [System.IO.File]::WriteAllText($config.OutputPrefix + '.err.log', $errTask.GetAwaiter().GetResult()) }
    $report = [ordered]@{
        Name=$config.Name;Executable=$config.Executable;Arguments=$config.Arguments;WorkingDirectory=$config.WorkingDirectory
        ExitCode=$exitCode;TimedOut=$timeout;ElapsedSeconds=[Math]::Round($runWatch.Elapsed.TotalSeconds,3)
        Processes=@($records.Values | Select-Object ProcessId,ParentProcessId,CreationDate,CommandLine)
        CleanedProcesses=@($cleanup)
    }
    $report | ConvertTo-Json -Depth 6 | Set-Content -LiteralPath ($config.OutputPrefix + '.json') -Encoding utf8NoBOM
    $process.Dispose()
    Write-Output ('END {0} exit={1} seconds={2:N1}' -f $config.Name,$exitCode,$runWatch.Elapsed.TotalSeconds)
}
exit $exitCode
